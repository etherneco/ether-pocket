cmake_minimum_required(VERSION 3.16)
project(EtherPower VERSION 0.1.0 LANGUAGES NONE)
install(FILES power.py service.py cli.py DESTINATION share/etherpower)
install(PROGRAMS packaging/etherpower DESTINATION bin)
install(FILES packaging/etherpower.service DESTINATION lib/systemd/system)
install(FILES packaging/com.etherneco.EtherPower1.conf DESTINATION share/dbus-1/system.d)
install(FILES packaging/com.etherneco.etherpower.policy DESTINATION share/polkit-1/actions)
install(FILES PowerPage.qml PowerMenu.qml powerclient.h powerclient.cpp DESTINATION share/etherpower/integration)
install(DIRECTORY tests/ DESTINATION share/etherpower/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/etherpower.rst ../docs/etherpower-audit-2026-10-19.rst DESTINATION share/doc/etherpower)
install(FILES distributor.json DESTINATION share/etherpower)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherpower)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "etherhardware (>= 0.1.0), python3, python3-dbus, python3-gi, systemd, dbus, polkitd")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/prerm")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
include(CPack)
