cmake_minimum_required(VERSION 3.21)
project(EtherPhone VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Network)
add_library(etherphone-client STATIC phoneclient.cpp phoneclient.h)
target_link_libraries(etherphone-client PUBLIC Qt6::Core Qt6::Gui Qt6::DBus)
target_include_directories(etherphone-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(etherphone main.cpp)
target_link_libraries(etherphone PRIVATE etherphone-client Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network)
qt_add_resources(etherphone ui PREFIX "/" FILES Main.qml CallSettings.qml)
install(TARGETS etherphone RUNTIME DESTINATION bin)
install(FILES core.py provider.py service.py cli.py DESTINATION share/etherphone)
install(PROGRAMS packaging/etherphone-control DESTINATION bin)
install(FILES packaging/etherphone.service DESTINATION lib/systemd/user)
install(FILES packaging/com.etherneco.EtherPhone1.service DESTINATION share/dbus-1/services)
install(FILES packaging/com.etherneco.EtherContacts1.service DESTINATION share/dbus-1/services)
install(FILES packaging/etherphone.desktop DESTINATION share/applications)
install(FILES CallSettings.qml phoneclient.cpp phoneclient.h distributor.json DESTINATION share/etherphone/integration)
install(DIRECTORY tests/ DESTINATION share/etherphone/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/etherphone-feasibility-2026-10-20.rst ../docs/etherphone.rst ../docs/etherphone-delivery-2026-10-20.rst DESTINATION share/doc/etherphone)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(etherphone-ui-test tests/ui-test.cpp)
 target_link_libraries(etherphone-ui-test PRIVATE etherphone-client Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Test)
 qt_add_resources(etherphone-ui-test testui PREFIX "/" FILES Main.qml CallSettings.qml IncomingOverlay.qml tests/overlay-test.qml)
 add_test(NAME etherphone-policy COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -v)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherphone)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, modemmanager, dbus-user-session, etheraudio (>= 0.1.0), ethermobile (>= 0.1.1), qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtquick-templates, qml6-module-qtqml-workerscript")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "EtherPhone native telephony engineering release; real calling and audio unverified")
include(CPack)
