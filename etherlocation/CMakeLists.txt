cmake_minimum_required(VERSION 3.16)
project(EtherLocation VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Sql)
find_package(PkgConfig REQUIRED)
pkg_check_modules(QMI REQUIRED IMPORTED_TARGET qmi-glib qrtr-glib)
add_executable(etherlocation-qmi qmi_gnss.cpp)
target_link_libraries(etherlocation-qmi PRIVATE Qt6::Core PkgConfig::QMI)
add_library(location-client STATIC locationclient.cpp locationclient.h)
target_link_libraries(location-client PUBLIC Qt6::Core Qt6::DBus)
add_executable(etherlocation-map map_main.cpp)
target_link_libraries(etherlocation-map PRIVATE location-client Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Sql)
qt_add_resources(etherlocation-map ui PREFIX "/" FILES Map.qml)
install(TARGETS etherlocation-qmi RUNTIME DESTINATION libexec)
install(TARGETS etherlocation-map RUNTIME DESTINATION bin)
install(FILES core.py identity.py service.py gnss.py maps.py providers.py client.py ip_provider.py DESTINATION share/etherlocation)
install(PROGRAMS packaging/etherlocation DESTINATION bin)
install(FILES packaging/etherlocation.service DESTINATION lib/systemd/user)
install(FILES packaging/com.etherneco.EtherLocation1.service DESTINATION share/dbus-1/services)
install(FILES packaging/etherlocation-map.desktop DESTINATION share/applications)
install(FILES LocationPage.qml locationclient.h locationclient.cpp DESTINATION share/etherlocation/integration)
install(DIRECTORY tests/ DESTINATION share/etherlocation/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/etherlocation.rst ../docs/etherlocation-feasibility-2026-10-20.rst distributor.json DESTINATION share/doc/etherlocation)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherlocation)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, dbus-user-session, x11-utils, libqt6sql6-sqlite, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, etherlock (>= 0.1.0), ethernotifications (>= 0.1.0), etherhardware (>= 0.1.0), etherpower (>= 0.1.0)")
include(CPack)
