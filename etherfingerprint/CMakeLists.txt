cmake_minimum_required(VERSION 3.16)
project(EtherFingerprint VERSION 0.2.0 LANGUAGES NONE)
install(FILES core.py service.py cli.py DESTINATION share/etherfingerprint)
install(PROGRAMS packaging/etherfingerprint DESTINATION bin)
install(FILES packaging/etherfingerprintd.service DESTINATION lib/systemd/system)
install(FILES packaging/etherfingerprint.conf DESTINATION lib/sysusers.d)
install(FILES packaging/com.etherneco.EtherFingerprint1.conf DESTINATION share/dbus-1/system.d)
install(FILES FingerprintPage.qml fingerprintclient.h fingerprintclient.cpp DESTINATION share/etherfingerprint/integration)
install(FILES distributor.json DESTINATION share/etherfingerprint)
install(FILES ../docs/etherfingerprint.rst ../docs/etherfingerprint-manager.rst DESTINATION share/doc/etherfingerprint)
install(DIRECTORY tests/ DESTINATION share/etherfingerprint/tests FILES_MATCHING PATTERN "*.py")
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherfingerprint)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, dbus, systemd")
set(CPACK_DEBIAN_PACKAGE_DESCRIPTION "Read-only EtherFingerprint diagnostics; biometric authentication blocked")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
include(CPack)
