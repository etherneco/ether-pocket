cmake_minimum_required(VERSION 3.21)
project(EtherNotifications VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
install(FILES core.py service.py DESTINATION share/ethernotifications)
install(FILES packaging/ethernotifications.service DESTINATION lib/systemd/user)
install(FILES packaging/org.freedesktop.Notifications.service packaging/com.etherneco.EtherNotifications1.service DESTINATION share/dbus-1/services)
install(FILES NotificationCenter.qml NotificationSettings.qml NotificationBanner.qml notificationclient.h notificationclient.cpp distributor.json DESTINATION share/ethernotifications/integration)
install(DIRECTORY tests/ DESTINATION share/ethernotifications/tests FILES_MATCHING PATTERN "*.py" PATTERN "*.qml" PATTERN "*.cpp")
install(FILES ../docs/ethernotifications.rst ../docs/ethernotifications-delivery-2026-10-20.rst DESTINATION share/doc/ethernotifications)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus)
 add_executable(ethernotifications-ui-test tests/ui-test.cpp)
 target_link_libraries(ethernotifications-ui-test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus)
 qt_add_resources(ethernotifications-ui-test ui PREFIX "/ethernotifications" FILES NotificationCenter.qml NotificationSettings.qml NotificationBanner.qml)
 qt_add_resources(ethernotifications-ui-test fixture PREFIX "/" FILES tests/ui-test.qml)
 add_executable(ethernotifications-client-test tests/client-test.cpp notificationclient.cpp notificationclient.h)
 target_include_directories(ethernotifications-client-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(ethernotifications-client-test PRIVATE Qt6::Core Qt6::DBus)
 add_test(NAME notification-policy COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_*.py -v)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethernotifications)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, dbus-user-session")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Persistent notification service and EtherShell integration; device wake and haptics unverified")
include(CPack)
