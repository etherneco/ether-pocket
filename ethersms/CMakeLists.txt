cmake_minimum_required(VERSION 3.21)
project(EtherSMS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Network Sql)
include(../ethermobile/Core.cmake)
add_library(ethersms-core STATIC repository.cpp repository.h controller.cpp controller.h speech.cpp speech.h settings.cpp settings.h)
target_link_libraries(ethersms-core PUBLIC ethermobile-core Qt6::Gui Qt6::Network Qt6::Sql)
target_include_directories(ethersms-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(ethersms-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ethersms main.cpp)
target_link_libraries(ethersms PRIVATE ethersms-core Qt6::Quick Qt6::Qml Qt6::QuickControls2)
qt_add_resources(ethersms ui PREFIX "/" FILES Main.qml ethersms.svg)
install(TARGETS ethersms RUNTIME DESTINATION bin)
install(FILES ethersms.desktop DESTINATION share/applications)
install(FILES ethersms.svg DESTINATION share/icons/hicolor/scalable/apps)
configure_file(ethersms-model ${CMAKE_CURRENT_BINARY_DIR}/ethersms-model @ONLY NEWLINE_STYLE UNIX)
install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/ethersms-model DESTINATION libexec/ethersms)
install(FILES README.rst DESTINATION share/doc/ethersms)
install(FILES settings.schema.json DESTINATION share/ethersms)
# Board integration is supplied for review; a generic app install never changes system authorization or audio.
install(FILES ../config/49-etherpocket-sms.rules ../config/audio/HiFi.conf DESTINATION share/ethersms/integration)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(ethersms-test tests.cpp)
 target_link_libraries(ethersms-test PRIVATE ethersms-core Qt6::Test Qt6::Quick Qt6::Qml Qt6::QuickControls2)
 qt_add_resources(ethersms-test test-ui PREFIX "/" FILES Main.qml ethersms.svg)
 add_executable(ethersms-live-test live-test.cpp)
 target_link_libraries(ethersms-live-test PRIVATE ethersms-core)
 # Never a CTest entry: requires an explicitly authorized real recipient.
 add_test(NAME ethersms-core COMMAND ethersms-test)
 set_tests_properties(ethersms-core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethersms)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Etherneco LTD")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
if(CMAKE_CROSSCOMPILING)
 set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "ethermobile (>= 0.1.1), libqt6sql6-sqlite, pulseaudio-utils, python3, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, qt6-svg-plugins")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "espeak-ng")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "EtherSMS native SMS with private local history and offline dictation integration")
include(CPack)
