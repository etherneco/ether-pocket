cmake_minimum_required(VERSION 3.16)
project(EtherMAP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Sql)
add_executable(ethermap main.cpp)
target_link_libraries(ethermap PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Sql)
option(ETHERMAP_UI_TESTS "Compile isolated native interaction acceptance driver" OFF)
if(ETHERMAP_UI_TESTS)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 target_compile_definitions(ethermap PRIVATE ETHERMAP_UI_TESTS)
 target_link_libraries(ethermap PRIVATE Qt6::Test)
endif()
qt_add_resources(ethermap ui PREFIX "/" FILES Main.qml)
install(TARGETS ethermap RUNTIME DESTINATION bin)
install(FILES backend.py domain.py network.py DESTINATION share/ethermap)
install(FILES ../build/ethermap-westminster-osm.mbtiles DESTINATION share/ethermap/maps RENAME westminster.mbtiles)
install(FILES ../build/ethermap-westminster.osm ../evidence/ethermap-osm-dataset.json DESTINATION share/doc/ethermap/osm)
install(FILES packaging/ethermap.desktop DESTINATION share/applications)
install(FILES distributor.json ../docs/ethermap.rst DESTINATION share/doc/ethermap)
install(DIRECTORY tests/ DESTINATION share/ethermap/tests FILES_MATCHING PATTERN "*.py")
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethermap)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "etherlocation (>= 0.1.1), python3, python3-dbus, python3-gi, libqt6sql6-sqlite, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates")
include(CPack)
