cmake_minimum_required(VERSION 3.16)
project(EtherUpdate VERSION 0.1.0 LANGUAGES NONE)
install(FILES security.py credentials.py provider.py artifacts.py history.py engine.py agent.py backup.py device.py service.py client.py DESTINATION share/etherupdate)
install(PROGRAMS packaging/etherupdate DESTINATION bin)
install(FILES packaging/etherupdate.service DESTINATION lib/systemd/system)
install(FILES packaging/com.etherneco.EtherUpdate1.conf DESTINATION share/dbus-1/system.d)
install(FILES packaging/com.etherneco.etherupdate.policy DESTINATION share/polkit-1/actions)
install(FILES packaging/policy.json packaging/trusted-keys.json DESTINATION /etc/etherupdate)
install(FILES UpdatePage.qml updateclient.cpp updateclient.h DESTINATION share/etherupdate/integration)
install(DIRECTORY tests/ DESTINATION share/etherupdate/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/etherupdate.rst DESTINATION share/doc/etherupdate)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherupdate)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-cryptography, python3-dbus, python3-gi, polkitd | policykit-1, systemd, apt, etherpower (>= 0.1.0), etherbackup (>= 0.1.0)")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/conffiles")
include(CPack)
