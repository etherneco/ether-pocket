cmake_minimum_required(VERSION 3.21)
project(EtherFiles VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent Test DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_library(ARCHIVE_LIBRARY NAMES archive libarchive.so.13 REQUIRED)
add_library(etherfiles-core STATIC src/files.cpp src/files.h src/operations.cpp src/operations.h src/services.cpp src/services.h)
target_link_libraries(etherfiles-core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent Qt6::DBus PkgConfig::GIO ${ARCHIVE_LIBRARY})
target_compile_options(etherfiles-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(etherfiles src/main.cpp)
target_link_libraries(etherfiles PRIVATE etherfiles-core Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_resources(etherfiles ui PREFIX "/" FILES qml/Main.qml qml/FilePane.qml qml/Tokens.qml qml/ActionButton.qml qml/TouchButton.qml qml/EntryField.qml qml/PlaceDelegate.qml icons/etherfiles.svg icons/folder.svg icons/file.svg)
qt_add_resources(etherfiles palette PREFIX "/shared" BASE "${CMAKE_CURRENT_SOURCE_DIR}/../ethershell/qml" FILES ../ethershell/qml/EtherPalette.qml)
install(TARGETS etherfiles RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/etherfiles.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES icons/etherfiles.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.rst DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/etherfiles)
install(FILES ../docs/etherfiles-2026-10-17.rst ../docs/etherfiles-network-2026-10-17.rst DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/etherfiles)
include(CTest)
if(BUILD_TESTING)
  add_executable(etherfiles-tests tests/core_test.cpp)
  target_link_libraries(etherfiles-tests PRIVATE etherfiles-core Qt6::Test)
  add_test(NAME filesystem-safety COMMAND etherfiles-tests)
  add_executable(etherfiles-network-tests tests/network_test.cpp)
  target_link_libraries(etherfiles-network-tests PRIVATE etherfiles-core Qt6::Test)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherfiles)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Etherneco LTD")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
if(CMAKE_CROSSCOMPILING)
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, libglib2.0-bin, gvfs, gvfs-backends, gvfs-fuse, openssh-client, udisks2, xdg-utils, qt6-svg-plugins")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "xfce4-terminal, poppler-utils, ffmpeg, gnome-keyring")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "EtherFiles native touch-first file manager for X11")
include(CPack)
