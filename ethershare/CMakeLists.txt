cmake_minimum_required(VERSION 3.21)
project(EtherShare VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus)
add_library(ethershare-client STATIC client.cpp client.h)
target_link_libraries(ethershare-client PUBLIC Qt6::Core Qt6::DBus)
target_include_directories(ethershare-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ethershare main.cpp)
target_link_libraries(ethershare PRIVATE ethershare-client Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_resources(ethershare ui PREFIX "/ethershare" FILES Main.qml SharePage.qml)
file(GLOB SETTINGS_QML RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../ethershell" "../ethershell/qml/settings/*.qml")
set(SETTING_FILES)
foreach(f ${SETTINGS_QML})
 list(APPEND SETTING_FILES "../ethershell/${f}")
endforeach()
qt_add_resources(ethershare settings PREFIX "/" BASE ../ethershell FILES ${SETTING_FILES})
install(TARGETS ethershare RUNTIME DESTINATION bin)
install(FILES ethershare.desktop DESTINATION share/applications)
install(FILES ethershare.svg DESTINATION share/icons/hicolor/scalable/apps)
foreach(f core.py nm.py service.py)
 configure_file(${f} ${CMAKE_CURRENT_BINARY_DIR}/${f} @ONLY NEWLINE_STYLE UNIX)
 install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${f} DESTINATION libexec/ethershare)
endforeach()
install(FILES packaging/ethershare.service DESTINATION lib/systemd/system)
install(FILES packaging/ethershare.tmpfiles DESTINATION lib/tmpfiles.d RENAME ethershare.conf)
install(FILES packaging/com.etherneco.EtherShare1.conf DESTINATION share/dbus-1/system.d)
install(FILES packaging/com.etherneco.ethershare.policy DESTINATION share/polkit-1/actions)
install(FILES packaging/49-etherpocket-sharing.rules DESTINATION share/ethershare/integration)
configure_file(../scripts/ethermobile-ipa-data.py ${CMAKE_CURRENT_BINARY_DIR}/ethermobile-ipa-data.py @ONLY NEWLINE_STYLE UNIX)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ethermobile-ipa-data.py DESTINATION share/ethershare/integration)
install(FILES README.rst DESTINATION share/doc/ethershare)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethershare)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
if(CMAKE_CROSSCOMPILING)
 set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
foreach(f postinst prerm postrm)
 configure_file(packaging/${f} ${CMAKE_CURRENT_BINARY_DIR}/${f} @ONLY NEWLINE_STYLE UNIX)
endforeach()
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_BINARY_DIR}/postinst;${CMAKE_CURRENT_BINARY_DIR}/prerm;${CMAKE_CURRENT_BINARY_DIR}/postrm")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, network-manager, dnsmasq-base, iptables, iproute2, curl, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native EtherPocket internet sharing with reversible routing and protected USB recovery")
include(CPack)
