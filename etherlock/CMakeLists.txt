cmake_minimum_required(VERSION 3.16)
project(EtherLock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
find_package(X11 REQUIRED)
add_executable(auth_etherlock auth.cpp)
qt_add_resources(auth_etherlock ui PREFIX "/" FILES Lock.qml)
target_link_libraries(auth_etherlock PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::QuickControls2 X11::X11 X11::Xi)
add_executable(etherlock-pam pam.cpp)
find_path(PAM_INCLUDE_DIR security/pam_appl.h PATHS /opt/etherpocket/arm64-sysroot/usr/include REQUIRED)
find_library(PAM_LIBRARY NAMES pam libpam.so.0 PATHS /usr/lib/aarch64-linux-gnu /opt/etherpocket/arm64-sysroot/usr/lib/aarch64-linux-gnu REQUIRED)
target_include_directories(etherlock-pam PRIVATE ${PAM_INCLUDE_DIR})
target_link_libraries(etherlock-pam PRIVATE Qt6::Core ${PAM_LIBRARY})
if(CMAKE_CROSSCOMPILING)
 target_link_options(etherlock-pam PRIVATE "-Wl,-rpath-link,/opt/etherpocket/arm64-sysroot/usr/lib/aarch64-linux-gnu")
endif()
install(TARGETS auth_etherlock etherlock-pam RUNTIME DESTINATION lib/etherlock)
install(FILES core.py service.py cli.py validate.py DESTINATION share/etherlock)
install(PROGRAMS packaging/etherlock packaging/etherlock-validate DESTINATION bin)
install(FILES lockclient.cpp lockclient.h LockPage.qml DESTINATION share/etherlock/integration)
install(FILES distributor.json DESTINATION share/etherlock)
install(DIRECTORY tests/ DESTINATION share/etherlock/tests FILES_MATCHING PATTERN "*.py")
install(FILES packaging/etherlock.pam DESTINATION /etc/pam.d RENAME etherlock)
install(FILES packaging/etherlock.service DESTINATION lib/systemd/user)
install(FILES ../docs/etherlock.rst DESTINATION share/doc/etherlock)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherlock)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS_PRIVATE_DIRS "/opt/etherpocket/arm64-sysroot/usr/lib/aarch64-linux-gnu")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/conffiles")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "xsecurelock (>= 1.9), libpam0g (>= 1.7), libpam-modules, python3, python3-dbus, python3-gi, libxss1, systemd, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates")
include(CPack)
