cmake_minimum_required(VERSION 3.16)
project(EtherHardware VERSION 0.1.1 LANGUAGES NONE)
install(FILES core.py backends.py service.py cli.py repair.py repair_entry.py DESTINATION share/etherhardware)
install(FILES vendor-profile.json DESTINATION share/etherhardware)
install(PROGRAMS packaging/etherhardware DESTINATION bin)
install(PROGRAMS packaging/etherhardware-repair DESTINATION libexec)
install(FILES packaging/etherhardware.service DESTINATION lib/systemd/system)
install(FILES packaging/com.etherneco.EtherHardware1.conf DESTINATION share/dbus-1/system.d)
install(FILES packaging/com.etherneco.etherhardware.policy DESTINATION share/polkit-1/actions)
install(FILES ../docs/etherhardware.rst ../docs/etherhardware-audit-2026-10-19.rst DESTINATION share/doc/etherhardware)
install(FILES HardwarePage.qml hardwareclient.h hardwareclient.cpp DESTINATION share/etherhardware/integration)

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etherhardware)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})

set(CPACK_PACKAGE_CONTACT "Etherneco maintainers")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, systemd, dbus, polkitd, pkexec, adduser, kmod")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/prerm")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
include(CPack)
