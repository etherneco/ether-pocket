cmake_minimum_required(VERSION 3.16)
project(EtherBluetooth VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Test)
add_executable(etherbluetooth main.cpp bluetooth.cpp bluetooth.h agent.cpp agent.h audio.cpp audio.h)
target_link_libraries(etherbluetooth PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus)
qt_add_resources(etherbluetooth ui PREFIX "/" FILES Main.qml bluetooth.svg)
install(TARGETS etherbluetooth RUNTIME DESTINATION bin)
install(PROGRAMS ../scripts/etherbluetooth-doctor.py DESTINATION libexec/etherbluetooth)
install(PROGRAMS ../scripts/etherbluetooth-recover.py DESTINATION libexec/etherbluetooth)
install(FILES etherbluetooth.desktop DESTINATION share/applications)
enable_testing()
add_executable(policy-test policy-test.cpp)
target_link_libraries(policy-test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME policy COMMAND policy-test)
add_executable(audio-policy-test audio-policy-test.cpp audio.cpp audio.h)
target_link_libraries(audio-policy-test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME audio-policy COMMAND audio-policy-test)
add_executable(integration-test integration-test.cpp bluetooth.cpp bluetooth.h agent.cpp agent.h)
target_link_libraries(integration-test PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "bluez, python3, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, qt6-svg-plugins, pulseaudio-utils")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "pkexec, pavucontrol")
include(CPack)

qt_add_resources(etherbluetooth structured-data PREFIX "/" BASE .. FILES ../qml/StructuredData.js)
