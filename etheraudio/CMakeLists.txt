cmake_minimum_required(VERSION 3.16)
project(EtherAudio VERSION 0.1.1 LANGUAGES NONE)
install(FILES core.py service.py cli.py DESTINATION share/etheraudio)
install(PROGRAMS packaging/etheraudio DESTINATION bin)
install(FILES packaging/etheraudio.service DESTINATION lib/systemd/user)
install(FILES packaging/com.etherneco.EtherAudio1.service DESTINATION share/dbus-1/services)
install(FILES AudioPage.qml audioclient.cpp audioclient.h DESTINATION share/etheraudio/integration)
install(DIRECTORY tests/ DESTINATION share/etheraudio/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/etheraudio.rst DESTINATION share/doc/etheraudio)
install(FILES distributor.json DESTINATION share/etheraudio)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME etheraudio)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, pulseaudio-utils, dbus-user-session, etherhardware (>= 0.1.0)")
include(CPack)
