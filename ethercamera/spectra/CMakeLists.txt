cmake_minimum_required(VERSION 3.16)
project(EtherCameraSpectra VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SPECTRA_UAPI_ROOT "" CACHE PATH "include/uapi directory of the matching running Qualcomm kernel")
foreach(header cam_defs.h cam_req_mgr.h cam_isp.h cam_sensor.h)
 if(NOT EXISTS "${SPECTRA_UAPI_ROOT}/media/${header}")
  message(FATAL_ERROR "Missing matching kernel UAPI: ${header}. Set SPECTRA_UAPI_ROOT.")
 endif()
endforeach()
find_package(Qt6 REQUIRED COMPONENTS Core)
# Isolate the vendor media UAPI. Do not replace libc's installed Linux
# headers with unsanitized in-kernel headers (linux/compiler.h etc.).
file(GLOB vendor_headers "${SPECTRA_UAPI_ROOT}/media/*.h")
foreach(header IN LISTS vendor_headers)
 get_filename_component(name "${header}" NAME)
 configure_file("${header}" "${CMAKE_CURRENT_BINARY_DIR}/vendor-uapi/media/${name}" COPYONLY)
endforeach()
add_library(spectra-transport STATIC transport.cpp)
target_include_directories(spectra-transport PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/vendor-uapi")
target_compile_options(spectra-transport PRIVATE -Wall -Wextra -Werror)
add_executable(ethercamera-spectra main.cpp)
target_link_libraries(ethercamera-spectra PRIVATE spectra-transport Qt6::Core)
add_executable(ethercamera-spectra-capture capture.cpp)
add_library(spectra-image STATIC image.cpp)
target_link_libraries(spectra-image PUBLIC Qt6::Core)
target_link_libraries(ethercamera-spectra-capture PRIVATE spectra-transport spectra-image Qt6::Core)
add_executable(spectra-image-test test_image.cpp)
target_link_libraries(spectra-image-test PRIVATE spectra-image)
target_compile_options(spectra-image-test PRIVATE -UNDEBUG)
add_executable(spectra-transport-test test_transport.cpp)
target_link_libraries(spectra-transport-test PRIVATE spectra-transport)
target_compile_options(spectra-transport-test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME lifecycle-and-errors COMMAND spectra-transport-test)
add_test(NAME raw10-and-tone COMMAND spectra-image-test)
# No automatic camera startup and no fabricated V4L2 capture endpoint.
install(TARGETS ethercamera-spectra ethercamera-spectra-capture RUNTIME DESTINATION libexec/ethercamera)
install(FILES module_profile.py capture_profile.py DESTINATION share/ethercamera/spectra)
install(FILES ../../docs/ethercamera-spectra-driver-2026-10-20.rst DESTINATION share/doc/ethercamera-spectra)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethercamera-spectra)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Qualcomm Spectra rear IMX586 RAW10 capture for the pinned GM1920 kernel")
include(CPack)
