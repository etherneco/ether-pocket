cmake_minimum_required(VERSION 3.16)
project(EtherCamera VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Multimedia)
add_executable(ethercamera-media media_main.cpp)
target_link_libraries(ethercamera-media PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Multimedia)
qt_add_resources(ethercamera-media media-ui PREFIX "/" FILES MediaLibrary.qml)
add_library(camera-controller STATIC capture_controller.cpp capture_controller.h)
target_link_libraries(camera-controller PUBLIC Qt6::Gui Qt6::Qml Qt6::Quick Qt6::DBus)
add_executable(ethercamera-app capture_main.cpp)
target_link_libraries(ethercamera-app PRIVATE camera-controller Qt6::QuickControls2)
qt_add_resources(ethercamera-app capture-ui PREFIX "/" FILES Capture.qml)
option(ETHERCAMERA_UI_TEST "Build the private-bus capture UI acceptance runner" OFF)
if(ETHERCAMERA_UI_TEST)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(ethercamera-ui-test tests/capture_ui_test.cpp)
 target_link_libraries(ethercamera-ui-test PRIVATE camera-controller Qt6::QuickControls2 Qt6::Test)
 qt_add_resources(ethercamera-ui-test capture-test-ui PREFIX "/" FILES Capture.qml)
endif()
install(TARGETS ethercamera-media ethercamera-app RUNTIME DESTINATION bin)
install(FILES packaging/ethercamera-media.desktop packaging/ethercamera-app.desktop DESTINATION share/applications)
install(FILES core.py devices.py provider.py spectra_provider.py documents.py audio.py power_guard.py capture_client.py location_metadata.py service.py cli.py DESTINATION share/ethercamera)
install(PROGRAMS packaging/ethercamera DESTINATION bin)
install(FILES packaging/ethercamera.service DESTINATION lib/systemd/user)
install(FILES packaging/com.etherneco.EtherCamera1.service DESTINATION share/dbus-1/services)
install(FILES CameraPage.qml cameraclient.h cameraclient.cpp DESTINATION share/ethercamera/integration)
install(DIRECTORY tests/ DESTINATION share/ethercamera/tests FILES_MATCHING PATTERN "*.py")
install(FILES ../docs/ethercamera.rst ../docs/ethercamera-feasibility-2026-10-20.rst ../docs/ethercamera-delivery-2026-10-20.rst ../docs/ethercamera-followup-2026-10-20.rst ../docs/ethercamera-app-2026-10-20.rst DESTINATION share/doc/ethercamera)
install(FILES distributor.json DESTINATION share/ethercamera)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethercamera)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, python3-dbus, python3-gi, gir1.2-gstreamer-1.0, gir1.2-gst-plugins-base-1.0, gstreamer1.0-plugins-base, gstreamer1.0-plugins-good, python3-pil, python3-zxing-cpp, python3-opencv, v4l-utils, dbus-user-session, etheraudio (>= 0.1.0), etherlock (>= 0.1.0), etherhardware (>= 0.1.0), etherpower (>= 0.1.0), qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtquick-templates, qml6-module-qtqml-workerscript, qml6-module-qtmultimedia")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "gstreamer1.0-plugins-ugly, gstreamer1.0-plugins-bad, ffmpeg, qml6-module-qtmultimedia")
include(CPack)
