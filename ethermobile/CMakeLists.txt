cmake_minimum_required(VERSION 3.21)
project(EtherMobile VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus)
include(Core.cmake)
add_executable(ethermobile main.cpp)
target_link_libraries(ethermobile PRIVATE ethermobile-core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_resources(ethermobile ui PREFIX "/" FILES Main.qml ethermobile.svg diagnostics.js)
install(TARGETS ethermobile RUNTIME DESTINATION bin)
install(PROGRAMS ../scripts/ethermobile-doctor.py DESTINATION libexec/ethermobile)
install(FILES ethermobile.desktop DESTINATION share/applications)
install(FILES ethermobile.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.rst ../docs/ethermobile-2026-10-18.rst DESTINATION share/doc/ethermobile)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(policy-test policy-test.cpp)
    target_link_libraries(policy-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME cellular-state-policy COMMAND policy-test)
    add_executable(profile-test profile-test.cpp)
    target_link_libraries(profile-test PRIVATE ethermobile-core Qt6::Test)
    add_executable(configuration-test configuration-test.cpp)
    target_link_libraries(configuration-test PRIVATE ethermobile-core Qt6::Test)
    # Deliberately not a CTest entry: opt-in test writes a real temporary NM profile.
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME ethermobile)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Etherneco LTD")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Etherneco LTD")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
if(CMAKE_CROSSCOMPILING)
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "modemmanager, network-manager (>= 1.44), python3, libqmi-utils, qrtr-tools, qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, qt6-svg-plugins")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "EtherMobile native cellular manager and read-only modem diagnostics")
include(CPack)
